  ∀ {a b c d e f x y : ℕ}
  → a ≤ b
  → c ≤ d
  → e ≤ f
  → suc
      (5 · ((b + d + f) ∸ (a + c + e))
       + 3 · ((b ∸ a) + (d ∸ c))
       + ((f ∸ e) + x) · (y + 2)
       + (a + c + e)
       + 7 · x)
    ≡
      (5 · ((b ∸ a) + ((d ∸ c) + (f ∸ e)))
       + 3 · ((d + b) ∸ (c + a))
       + ((f + x) ∸ e) · (2 + y)
       + e + c + a
       + x + 6 · x) + 1
